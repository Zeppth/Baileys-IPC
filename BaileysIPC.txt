   5 |      * Manages asynchronous request-response cycles between main thread and worker threads via IPC.
   6 |      * @param {Function} sendCallback - Function used to dispatch messages to the worker.
   7 |      */

   9 |         this.requests = new Map();
  10 |         this.send = sendCallback;
  11 |     }
  12 | 
  13 |     /**
  14 |      * Sends an IPC request and returns a promise that awaits the worker's response.
  15 |      * @param {Object} object - The payload to send.
  16 |      * @param {number} [timeout=10000] - Timeout duration in milliseconds.
  17 |      * @returns {Promise<any>} Response promise.
  18 |      */
  19 |     send(object, timeout = 10000) {
  20 |         const requestId = generateId();
  21 |         return new Promise((resolve, reject) => {
  22 | 
  23 |             const timer = timeout ? setTimeout(() => {
  24 |                 if (!this.requests.has(requestId)) return;
  25 |                 this.reject(requestId, `Timeout [${requestId}]`);
  26 |             }, timeout) : null;
  27 | 
  28 |             this.requests.set(requestId, { resolve, reject, timer });
  29 | 
  30 |             if (typeof this.send === 'function') {
  31 |                 this.send({ ...object, requestId });
  32 |             }
  33 |         });
  34 |     }
  35 | 
  36 |     /**
  37 |      * Resolves a pending IPC request by its unique ID.
  38 |      * @param {string} requestId - The unique identifier of the request.

  40 |      * @returns {boolean} True if successfully resolved, false otherwise.
  41 |      */

  43 |         if (!this.requests.has(requestId)) return false;
  44 |         const { resolve, timer } = this.requests.get(requestId);
  45 |         if (timer) clearTimeout(timer); this.requests.delete(requestId);
  46 |         resolve(data); return true;
